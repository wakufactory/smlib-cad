# Concave XY section, extruded symmetrically about its plane.
profile bracket_section
  polygon 0 0  18 0  18 0  18 5  6 5  6 16  0 16
end
extrude bracket bracket_section -28 -8 0 -3 3
material bracket 0.12 0.55 0.85 0.5 0.3

# Positive draft narrows the top face; the same profile can still be extruded normally.
profile taper_section
  polygon 0 0  10 0  10 8  0 8
end
extrude tapered_block taper_section 92 0 0 0 5 draft 8
material tapered_block 0.9 0.65 0.2 0.4 0.3

# Radius/height section: a closed annular sleeve after a full revolution.
profile wall
  polygon 5 0  9 0  9 12  5 12
end
revolve sleeve wall 0 0 -6 0 360
material sleeve 0.85 0.35 0.12 0.65 0.3

# Partial revolution: start/end faces are closed. Axis +X, zero-angle ray +Y.
revolve sector wall 24 0 -6 30 270 1 0 0 0 1 0
material sector 0.45 0.75 0.25 0.3 0.45

# One reusable curved section with a corner for both extrusion and revolution.
profile curved_section
  spline through 2 0  4 0  4 0  5 1 \
    5 3  4 4  2 4  1 3  1 1 closed
end
extrude curved_plate curved_section 48 0 0 0 2
material curved_plate 0.75 0.25 0.55 0.4 0.3
revolve curved_ring curved_section 72 0 0 0 360
material curved_ring 0.25 0.72 0.7 0.5 0.28
