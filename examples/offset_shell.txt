# Offset every face outward; box edges remain sharp.
box expanded -24 -5 0 10 10 10
offset expanded 1
material expanded 0.15 0.55 0.85 0.4 0.3

# face 1 is the +Z face of this freshly created box (see faces output).
box cup -5 -5 0 10 10 10
faces cup
shell cup -1 1
material cup 0.85 0.35 0.12 0.5 0.3

# Remove two opposite faces to produce a rectangular tube.
box tube 14 -5 0 10 10 10
shell tube -1 0 1
material tube 0.45 0.75 0.25 0.3 0.45
